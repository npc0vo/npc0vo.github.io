##### Class f

这个类的核心逻辑是：

1. 通过 `b()` 方法解密得到需要管理的组件类名。 com.sagittarius.cr01.A
    
2. 以单例模式创建 `f` 实例，在初始化时通过反射实例化解密得到的组件。
    
3. 作为中间层，将自身收到的生命周期事件（如 `onCreate`、`attachBaseContext` 等）转发给所有管理的组件，实现组件的统一管理和事件分发。
    

##### Class g

这个类的核心作用是**通过反射操作 Android 框架的核心类（如****`ActivityThread`****），替换或修复****`Application`****实例及其上下文关联**，确保应用在不同设备（尤其是三星）上能正常运行。主要场景包括：

- 修复`StubApplication`的真实应用上下文；
    
- 替换系统中的`Application`实例，适配特殊设备逻辑；
    
- 绑定上下文到`Application`，保证资源访问正常。
    

  

简单就是调用libbaiduprotect.so 的n1 函数 获取解密后的dex 然后java使用 inMemoryDexLoader进行加载 并且替换Application为xiaomicar 是加密壳,不是抽取壳

```Python
import base64

def custom_base64_decode(s):
    """自定义Base64解码，模拟Java中的a.a()方法"""
    # 移除末尾的空白字符和等号
    s = s.rstrip(' \t\n\r=')
    
    # 标准Base64字符映射表
    base64_chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/="
    
    # 处理特殊字符映射（根据Java代码）
    s = s.replace('-', '+').replace('_', '/')
    
    # 计算需要补充的等号数量
    padding = (4 - (len(s) % 4)) % 4
    s += '=' * padding
    
    try:
        return base64.b64decode(s)
    except:
        return None

def decstr(s, s1):
    """解密函数，模拟Java中的decstr方法"""
    if not s:  # 空字符串直接返回
        return ""
    
    # 先进行Base64解码
    arr_b = custom_base64_decode(s)
    if arr_b is None:
        return ""
    
    # 异或解密
    key_chars = list(s1)
    key_len = len(key_chars)
    result = []
    
    for i, b in enumerate(arr_b):
        key_char = key_chars[i % key_len]
        # 异或运算并转换为字符
        result.append(chr(b ^ ord(key_char)))
    
    return ''.join(result)

if __name__ == "__main__":
    # 待解密的字符串和对应的密钥
    encrypted_strings = [
        ("BBsJaxcIFg0pH0kVFDUTAwdKLAkIGAUxHwM=", "gtdEz"),
        ("DgQ2OzYGDnwoKR9EEyotBhw7PSA7AiAsOAs=", "ojRIY"),
        ("KykpESYuKwoXKDQuPgA1", "BGZeG"),
        ("", "xTdRV")
    ]
    
    # 解密并输出结果
    for idx, (s, key) in enumerate(encrypted_strings):
        decrypted = decstr(s, key)
        print(f"解密结果 {idx}: {decrypted}")
```

def decrypt(cipher, keybox):
    # 计算密文长度的一半作为结果缓冲区大小
    v4 = len(cipher) // 2
    result = [0] * (v4 + 1)  # 预留终止符位置
    result[v4] = 0  # 终止符

    if not cipher:  # 处理空输入
        return bytes(result[:v4])

    n0x3A = cipher[0]
    n8 = 0
    v8_idx = 2  # v8 初始指向 cipher + 2
    v9_idx = 0  # 结果数组索引

    while n0x3A != 0:
        # 获取当前处理的两个字符
        n0x3A_1 = cipher[v8_idx - 1]

        # 计算v11
        if n0x3A >= 0x3A:
            v11 = -112
        else:
            v11 = 0

        # 计算v12
        if n0x3A_1 >= 0x3A:
            v12 = -55
        else:
            v12 = -48

        # n8循环（0-7）
        if n8 == 8:
            n8 = 0

        # 计算解密字节并异或
        decoded_byte = keybox[n8] ^ (v11 + 16 * n0x3A + v12 + n0x3A_1)
        # 确保结果为无符号字节
        result[v9_idx] = decoded_byte & 0xFF

        # 更新索引和变量
        v9_idx += 1
        n0x3A = cipher[v8_idx] if v8_idx < len(cipher) else 0
        n8 += 1
        v8_idx += 2

    return bytes(result[:v9_idx])

def get_aarch64_first_arg(call_ea):
    """
    针对 AArch64 架构，获取函数调用时的第一个参数 (X0 寄存器) 的值。
    通过向上追溯来查找在 BL 指令前设置 X0 的指令。

    :param call_ea: BL (Branch with Link) 指令的地址。

    ARG_REG = "X0"

    ea = idc.prev_head(call_ea)

    # 在函数开始前或向上追溯太多指令后停止
    func_start = idc.get_func_attr(call_ea, idc.FUNCATTR_START)

    # 安全地向上追溯一小段距离（例如 10 条指令）
    max_lookback = 10

    for i in range(max_lookback):

            return None

        # 获取指令助记符 (e.g., ADRL, MOV, LDR)

        if op_type0 == idc.o_reg and op_text0 == ARG_REG:

            op_type1 = idc.get_operand_type(ea, 1)

                # if value_name:

            elif mnem == "LDR" and op_type1 == idc.o_displ:
