```Python
#!/usr/bin/env python3
from pwn import *
context(os='linux',arch = 'amd64')
context.log_level = 'error'
filename = "./re/hyperjump"
elf = context.binary = ELF(filename)
gs = '''
b *$rebase(0x13f3)
'''

def start():
    if args.GDB:
        return gdb.debug(elf.path, gdbscript = gs, api = True)
    else:
        return process(elf.path)

def debug_program(input_data: str):

    io = start()
    vals = []
    gdb=io.gdb
    io.sendline(input_data)

    for i in range(25):
        sleep(0.1)
        vals.append(hex(int(gdb.parse_and_eval("$al"))&0xff))
        gdb.execute("c")
        sleep(0.1)

    #print(vals)

    io.close()
    return vals

from string import printable
cipher=[0x00, your_cipher] #自己补充
if __name__ == "__main__":
    #pos=printable.find('~')+1
    pos=0
    for ch in printable[pos:]:
        test_input=ch*24
        al_vals=debug_program(test_input)
        print("======")
        print(test_input)
        print(al_vals)

    print("Done!")
```

```Python
#!/usr/bin/env python3
"""
解析 crack.log 并根据说明：
，从下标为 1 开始，用 cipher 的值去匹配每个条目数组在相同下标位置的值。
 若相等，则该条目对应的重复字符就是该密文字节的明文字符。

用法: python3 recover_cipher.py [path/to/crack.log]
如果不提供路径，会默认读取 ./crack.log
"""
import re
import sys
import ast
from collections import defaultdict

def parse_log(path):
    with open(path, 'r', encoding='utf-8', errors='ignore') as f:
        data = f.read()

    # 分割每个块（以 =======... 分隔）
    blocks = [b.strip() for b in re.split(r"=+\n", data) if b.strip()]

    entries = []  # list of (label_char, list_of_ints)
    cipher = None

    # 逐块解析。块通常结构：一行是重复字符（例如 000... 或 aaaa...），下一行是数组字面量
    for b in blocks:
        # 尝试找到 'cipher:' 行
        m_cipher = re.search(r"cipher\s*:\s*\[([^\]]*)\]", b, re.IGNORECASE)
        if m_cipher:
            inner = m_cipher.group(1)
            # 将 hex/dec 元素解析为整数
            parts = [p.strip() for p in inner.split(',') if p.strip()]
            try:
                cipher = [int(p, 0) for p in parts]
            except Exception:
                # 有时元素如 0x4B 后面有逗号和空格，int(...,0) 仍能处理
                cipher = []
                for p in parts:
                    try:
                        cipher.append(int(p, 0))
                    except Exception:
                        # 忽略解析失败项
                        pass
            continue

        # 否则尝试提取第一行标签和后面的数组字面量
        lines = [l.strip() for l in b.splitlines() if l.strip()]
        if len(lines) < 2:
            continue
        label = lines[0]
        # 找到第一行类似数组的行
        arr_line = None
        for ln in lines[1:]:
            if ln.startswith('[') and ln.endswith(']'):
                arr_line = ln
                break
            # 有时数组会被打印为 Python 风格的字符串分多行；尝试搜索包含多个 0x 的行
            if '0x' in ln and ',' in ln:
                arr_line = ln
                break

        if not arr_line:
            continue

        # 以安全的方式把 arr_line 解析为一个列表
        try:
            arr = ast.literal_eval(arr_line)
        except Exception:
            # 尝试手动解析逗号分隔的 0x.. 内容
            parts = [p.strip().strip("'\"") for p in arr_line.strip('[]').split(',') if p.strip()]
            arr = []
            for p in parts:
                try:
                    arr.append(int(p, 0))
                except Exception:
                    # 忽略无法解析的
                    pass

        # 把可能的 hex 字符串转换为整数
        parsed = []
        for x in arr:
            if isinstance(x, str):
                try:
                    parsed.append(int(x, 0))
                except Exception:
                    # 如果是单字符字符串（label 内容），跳过
                    pass
            elif isinstance(x, int):
                parsed.append(x)
        if parsed:
            # 如果 label 是重复字符序列，取第一个字符作为该条目的字符
            ch = label[0] if label else '?'
            entries.append((ch, parsed))

    return entries, cipher

def recover(entries, cipher):
    if not cipher:
        print('未找到 cipher 数据。')
        return

    # 从下标 1 开始
    plaintext_chars = []
    details = {}

    max_index = len(cipher)
    for idx in range(1, max_index):
        target = cipher[idx]
        matches = []
        for ch, arr in entries:
            if idx < len(arr) and arr[idx] == target:
                matches.append(ch)
        if matches:
            # 如果存在多个匹配，我们取第一个并标注多个可能
            chosen = matches[0]
            plaintext_chars.append(chosen)
            details[idx] = {'target': target, 'matches': matches}
        else:
            plaintext_chars.append('?')
            details[idx] = {'target': target, 'matches': []}

    # 合成字符串
    plaintext = ''.join(plaintext_chars)
    return plaintext, details

def main():
    path = sys.argv[1] if len(sys.argv) > 1 else './crack.log'
    entries, cipher = parse_log(path)
    print(f'解析到 {len(entries)} 条条目；cipher 长度 = {len(cipher) if cipher else 0}')
    plaintext, details = recover(entries, cipher)

    print('\n逐下标匹配结果（从下标 1 开始）：')
    for idx in sorted(details.keys()):
        info = details[idx]
        tgt = info['target']
        matches = info['matches']
        if matches:
            print(f'index {idx}: 0x{tgt:02X} -> chars {matches}')
        else:
            print(f'index {idx}: 0x{tgt:02X} -> NO MATCH')

    print('\n合成明文（从下标1开始，未找到处为 ?）：')
    print(plaintext)

if __name__ == '__main__':
    main()
```

```Python
def decrypt_char(c):
    """解密单个字符"""
    if c.isupper():
        # 大写字母：解密公式 (15*(y - 11)) % 26
        y = ord(c) - ord('A')
        transformed = (15 * (y - 11)) % 26
        return chr(transformed + ord('A'))
    elif c.islower():
        # 小写字母：解密公式 (15*(y - 11)) % 26
        y = ord(c) - ord('a')
        transformed = (15 * (y - 11)) % 26
        return chr(transformed + ord('a'))
    elif c.isdigit():
        # 数字：解密公式 (3*(y - 11)) % 10
        y = ord(c) - ord('0')
        transformed = (3 * (y - 11)) % 10
        return chr(transformed + ord('0'))
    else:
        # 其他字符不解密
        return c

def decrypt(s):
```

```Plain
static void swap_bytes(uint8_t *a, uint8_t *b) {
    uint8_t temp = *a;
    *a = *b;
    *b = temp;
}

/**
 * @brief 还原 sub_401840 算法 (修改版的 RC4 in CFB-8 Mode)
 *
 * @param sbox       一个256字节的RC4 S-Box，必须是预先初始化过的。
 * @param data_ptr   指向要加密或解密的数据缓冲区的指针。
 * @param length     要处理的数据长度 (原始代码中的 i_1)。
 * @param inner_loop 内部循环的次数，用于生成每个密钥流字节。
 *
 * @return 始终返回 0xFADA21A8LL (与原始代码一致)。
 */
uint64_t custom_rc4(uint8_t *sbox, uint8_t *data_ptr, int length, int inner_loop) {
    int i, j;
    int rc4_i = 0;          // 对应原始代码的 I
    int rc4_j = 0;          // 对应原始代码的 j_1
    uint8_t keystream_byte; // 对应原始代码的 v7
    uint8_t feedback_byte;  // 对应原始代码的 v6
    
    // 1. 初始化反馈字节 (IV)
    // v6 = *sbox;
    feedback_byte = sbox[0];
    
    // 2. 初始化RC4指针
    // j_1 = 0;
    // I = 0;
    rc4_i = 0;
    rc4_j = 0;

    // 3. 主循环，处理每个数据字节
    // for ( i = 0; i < i_1; ++i )
    for (i = 0; i < length; ++i) {
        
        // 4. 内部循环，生成一个密钥流字节 v7
        // for ( j = 0; j < inner_loop; ++j )
        for (j = 0; j < inner_loop; ++j) {
            
            // 4a. RC4 PRGA 指针更新 I
            // I = (I + 1) % 256;
            rc4_i = (rc4_i + 1) % 256;
            
            // 4b. RC4 PRGA 指针更新 j_1
            // j_1 = ((unsigned __int8)sbox[I] + j_1) % 256;
            rc4_j = (sbox[rc4_i] + rc4_j) % 256;

            // 4c. S-Box 交换 (这是RC4 KSA中的一步, 被混入PRGA)
            // ((void...))(&sbox[I], &sbox[j_1]); // SWAP
            swap_bytes(&sbox[rc4_i], &sbox[rc4_j]);
            
            // 4d. 生成密钥流字节 v7
            // v7 = sbox[((unsigned __int8)sbox[j_1] + (unsigned __int8)sbox[I]) % 256];
            keystream_byte = sbox[(sbox[rc4_i] + sbox[rc4_j]) % 256];
        }
        
        // 5. CFB-8 模式加密/解密
        // *(_BYTE *)(input + i) ^= v6 ^ v7;
        data_ptr[i] ^= feedback_byte ^ keystream_byte;
        
        // 6. 更新反馈字节 v6
        // v6 = *(_BYTE *)(input + i);
        feedback_byte = data_ptr[i];
    }

    // 原始函数返回一个固定的 __int64 值
    return 0xFADA21A8LL;
}
/**
 * @brief 解密函数 (CFB-8 模式解密)
 */
uint64_t custom_rc4_decrypt(uint8_t *sbox, uint8_t *data_ptr, int length, int inner_loop) {
    int i, j;
    int rc4_i = 0;
    int rc4_j = 0;
    uint8_t keystream_byte;
    uint8_t feedback_byte;
    uint8_t temp_ciphertext;  // 关键:保存密文字节
    
    // 1. 初始化反馈字节 (IV)
    feedback_byte = sbox[0];
    
    // 2. 初始化RC4指针
    rc4_i = 0;
    rc4_j = 0;

    // 3. 主循环
    for (i = 0; i < length; ++i) {
        
        // 4. 内部循环,生成密钥流字节
        for (j = 0; j < inner_loop; ++j) {
            rc4_i = (rc4_i + 1) % 256;
            rc4_j = (sbox[rc4_i] + rc4_j) % 256;
            swap_bytes(&sbox[rc4_i], &sbox[rc4_j]);
            keystream_byte = sbox[(sbox[rc4_i] + sbox[rc4_j]) % 256];
        }
        
        // 5. 解密: 先保存密文,再解密,最后用密文更新反馈
        temp_ciphertext = data_ptr[i];           // 保存密文
        data_ptr[i] ^= feedback_byte ^ keystream_byte;  // 解密
        feedback_byte = temp_ciphertext;         // 反馈使用密文而非明文!
    }

    return 0xFADA21A8LL;
}
```

```JavaScript
function decrypt(cipher, key) {
    const keyCodes = Array.from(key).map(c => c.charCodeAt(0));
    const keyLen = keyCodes.length;
    const sum = keyCodes.reduce((acc, val, idx) => {
        switch (idx % 4) {
            case 0: return acc + 1 * val;
            case 1: return acc + val + 0;
            case 2: return acc + (0 | val);
            case 3: return acc + (0 ^ val);
            default: return acc;
        }
    }, 0);
    const c = sum % 8;

    let plainCodes = [];
    for (let o = 0; o < cipher.length; o++) {
        const h = cipher[o];
        let u;
        
        // 逆向位移操作
        switch (c) {
            case 0: u = h; break;
            case 1: u = (h >> 1) | ((h & 1) << 7); break;
            case 2: u = (h >> 2) | ((h & 3) << 6); break;
            case 3: u = (h >> 3) | ((h & 7) << 5); break;
            case 4: u = (h >> 4) | ((h & 15) << 4); break;
            case 5: u = (h >> 5) | ((h & 31) << 3); break;
            case 6: u = (h >> 6) | ((h & 63) << 2); break;
            case 7: u = (h >> 7) | ((h & 127) << 1); break;
        }

        // 逆向异或操作
        const keyCode = keyCodes[o % keyLen];
        let plainCode;
        switch (o % 3) {
            case 0: plainCode = u ^ keyCode; break;
            case 1: plainCode = u ^ keyCode; break;
            case 2: plainCode = u ^ keyCode; break;
        }
        plainCodes.push(plainCode);
    }
    return String.fromCharCode(...plainCodes);
}
```

```Python
import os

def decrypt(global_meta_data, global_meta_data_size):
    # 转换为无符号16位整数数组（模拟C中的unsigned __int16*）
    # 注意：Python中直接直接操作uint16数组，这里通过字节索引计算模拟
    meta_bytes = global_meta_data
    
    # 读取a的值（对应global_meta_data[512]，每个元素是2字节）
    a = int.from_bytes(meta_bytes[512*2 : 512*2 + 2], byteorder='little', signed=False)
    
    v4 = global_meta_data_size - 4 * a
    dest_size = v4 - 4
    dest = bytearray(dest_size)
    
    # 复制前0x400字节（1024字节）
    dest[:0x400] = meta_bytes[:0x400]
    
    size = v4 - 1028
    if size >= 1:
        # 加密数据起始位置（字节偏移）：global_meta_data[2*a + 514]对应的字节位置
        encrypted_start = (2 * a + 514) * 2  # 每个元素2字节
        
        # 循环处理每个4字节块
        for idx in range(0, size, 4):
            tmp = idx  # 原逻辑中idx >=0时tmp恒为idx
            
            # 计算v9和模运算结果
            v9 = idx + (idx // a)
            mod_result = v9 % a
            
            # 源数据位置（字节偏移）
            src_pos = encrypted_start + tmp
            # 密钥位置（对应global_meta_data[2*(v9%a) + 514]的字节偏移）
            key_pos = (2 * mod_result + 514) * 2
            
            # 读取4字节数据（注意边界处理）
            src_data = meta_bytes[src_pos : src_pos + 4]
            key_data = meta_bytes[key_pos : key_pos + 4]
            
            # 异或解密（不足4字节的部分按实际长度处理）
            decrypted = bytes(s1 ^ s2 for s1, s2 in zip(src_data, key_data))
            
            # 写入目标缓冲区
            dest_pos = 1024 + tmp  # 0x400 = 1024
            dest[dest_pos : dest_pos + len(decrypted)] = decrypted
    
    return bytes(dest)
```

```C++
void TeaEncrypt(uint32_t *v, uint32_t *key) {
    if (!v || !key) {
        return;
    }

    uint32_t v0 = v[0];
    uint32_t v1 = v[1];
    uint32_t sum = 0;
    uint32_t delta = 0x9E3779B9;

    for (int i = 0; i < 16; i++) {
        sum += delta;
        v0 += ((v1 << 4) + key[0]) ^ (v1 + sum) ^ ((v1 >> 5) + key[1]);
        v1 += ((v0 << 4) + key[2]) ^ (v0 + sum) ^ ((v0 >> 5) + key[3]);
    }

    v[0] = v0;
    v[1] = v1;
}

        delta -= 0x9E3779B9; // delta 递减

        // 解密 v1
        v1 -= ((v0 << 4) + k2) ^ (v0 + delta) ^ ((v0 >> 5) + k3);

        uint32_t k0 = key[0];
        uint32_t k1 = key[1];

        // 解密 v0
        v0 -= ((v1 << 4) + k0) ^ (v1 + delta) ^ ((v1 >> 5) + k1);
    }

    v[0] = v0;
    v[1] = v1;
}

// 小端序转换
uint32_t le_to_uint32(uint8_t *bytes) {
    return (bytes[0] << 0) | (bytes[1] << 8) | (bytes[2] << 16) | (bytes[3] << 24);
}

// uint32转小端序字节
void uint32_to_le(uint32_t val, uint8_t *bytes) {
    bytes[0] = (val >> 0) & 0xFF;
    bytes[1] = (val >> 8) & 0xFF;
    bytes[2] = (val >> 16) & 0xFF;
    bytes[3] = (val >> 24) & 0xFF;
}

// 打印明文字符串（如果可打印）
void print_plaintext(uint8_t *data, int len) {
    printf("Plaintext (ASCII): ");
    for (int i = 0; i < len; i++) {
        if (data[i] >= 32 && data[i] <= 126) { // 可打印字符
            printf("%c", data[i]);
        } else {
            printf("\\x%02X", data[i]);
        }
    }
    printf("\n");
}
```
