def rol32(x, n):
    """32位循环左移"""
    n = n % 32
    return ((x << n) | (x >> (32 - n))) & 0xFFFFFFFF

def calculate_R0_R1_formula(C0, C1):
    """
    根据公式直接计算R0和R1
    """
    # 预计算常量
    const_aa = 0x1010101 * 0xaa  # 0xaaaaaaaa
    const_bb = 0x1010101 * 0xbb  # 0xbbbbbbbb
    const_cc = 0x1010101 * 0xcc  # 0xcccccccc
    const_dd = 0x1010101 * 0xdd  # 0xdddddddd
    
    # 中间计算步骤
    temp1 = rol32(C1, 3)                    # C1 <<< 3
    temp2 = C0 ^ (const_aa ^ temp1)         # C0 ^ (0xaaaaaaaa ^ (C1 <<< 3))
    temp3 = rol32(temp2, 3)                 # (C0 ^ (0xaaaaaaaa ^ (C1 <<< 3))) <<< 3
    temp4 = C1 ^ (const_bb ^ temp3)         # C1 ^ ((C0 ^ (0xaaaaaaaa ^ (C1 <<< 3))) <<< 3) ^ 0xbbbbbbbb
    temp5 = rol32(temp4, 3)                 # (C1 ^ ((C0 ^ (0xaaaaaaaa ^ (C1 <<< 3))) <<< 3) ^ 0xbbbbbbbb) <<< 3
    
    # 计算R1
    R1 = temp2 ^ (const_cc ^ temp5)         # R1 = temp2 ^ (0xcccccccc ^ temp5)
    
    # 计算R0
    temp6 = rol32(R1, 3)                    # R1 <<< 3
    R0 = temp4 ^ (const_dd ^ temp6)         # R0 = temp4 ^ (0xdddddddd ^ temp6)
    
    return R0, R1

def encrypt_block(C0, C1):
    """
    加密一个数据块
    C0, C1: 输入的32位数据
    返回: (R0, R1) 加密后的32位数据
    """
    const_aa = 0xaaaaaaaa
    const_bb = 0xbbbbbbbb
    const_cc = 0xcccccccc
    const_dd = 0xdddddddd
    
    temp1 = rol32(C1, 3)
    temp2 = C0 ^ (const_aa ^ temp1)
    temp3 = rol32(temp2, 3)
    temp4 = C1 ^ (const_bb ^ temp3)
    temp5 = rol32(temp4, 3)
    R1 = temp2 ^ (const_cc ^ temp5)
    temp6 = rol32(R1, 3)
    R0 = temp4 ^ (const_dd ^ temp6)
    
    return R0, R1

def decrypt_block(R0, R1):
    """
    解密函数
    R0, R1: 加密后的32位数据
    返回: (C0, C1) 解密后的原始数据
    """
    const_aa = 0xaaaaaaaa
    const_bb = 0xbbbbbbbb
    const_cc = 0xcccccccc
    const_dd = 0xdddddddd
    
    # 解密步骤（加密的逆过程）
    
    # 步骤8的逆: R0 = temp4 ^ (0xdddddddd ^ temp6)
    # => temp4 = R0 ^ (0xdddddddd ^ temp6)
    # 但我们需要先求temp6
    temp6 = rol32(R1, 3)  # temp6 = R1 <<< 3
    
    # 现在可以求temp4
    temp4 = R0 ^ (const_dd ^ temp6)
    
    # 步骤6的逆: R1 = temp2 ^ (0xcccccccc ^ temp5)
    # => temp2 = R1 ^ (0xcccccccc ^ temp5)
    # 但我们需要先求temp5
    temp5 = rol32(temp4, 3)  # temp5 = temp4 <<< 3
    
    # 现在可以求temp2
    temp2 = R1 ^ (const_cc ^ temp5)
    
    # 步骤4的逆: temp4 = C1 ^ (0xbbbbbbbb ^ temp3)
    # => C1 = temp4 ^ (0xbbbbbbbb ^ temp3)
    # 但我们需要先求temp3
    temp3 = rol32(temp2, 3)  # temp3 = temp2 <<< 3
    
    # 现在可以求C1
    C1 = temp4 ^ (const_bb ^ temp3)
    
    # 步骤2的逆: temp2 = C0 ^ (0xaaaaaaaa ^ temp1)
    # => C0 = temp2 ^ (0xaaaaaaaa ^ temp1)
    # 但我们需要先求temp1
    temp1 = rol32(C1, 3)  # temp1 = C1 <<< 3
    
    # 现在可以求C0
    C0 = temp2 ^ (const_aa ^ temp1)
    
    return C0, C1
